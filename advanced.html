Advanced Concepts

These are the topics that separate junior developers from senior ones.

Hoisting & Temporal Dead Zone (TDZ)

Hoisting moves declarations to the top of their scope during compilation. 

console.log(a); // undefined (var is hoisted and initialized)
var a = 5;

console.log(b); // ReferenceError: Cannot access 'b' before initialization
let b = 10;

// The TDZ is the zone between the start of the block and the let/const declaration.
// During the TDZ, the variable exists but cannot be accessed.

The Prototype Chain

const obj = {};
console.log(obj.toString); // Works! Inherited from Object.prototype

// Check the prototype
Object.getPrototypeOf(obj) === Object.prototype; // true

// Create an object with no prototype
const dict = Object.create(null);
console.log(dict.toString); // undefined

Generators

Generators are functions that can pause and resume. They return an iterator. 

function* countUp(max) {
  for (let i = 0; i <= max; i++) {
    yield i; // Pause here, return i
  }
}

const gen = countUp(3);
console.log(gen.next()); // { value: 0, done: false }
console.log(gen.next()); // { value: 1, done: false }
console.log(gen.next()); // { value: 2, done: false }
console.log(gen.next()); // { value: 3, done: false }
console.log(gen.next()); // { value: undefined, done: true }

// Generators are iterable
for (const n of countUp(5)) {
  console.log(n);
}

Symbols

Symbols are unique, immutable primitive values. They are great for creating private-ish object keys. 

const id = Symbol('id');
const user = { name: "Ada", [id]: 123 };

console.log(user[id]); // 123
console.log(Object.keys(user)); // ["name"] — Symbol keys are hidden
console.log(Object.getOwnPropertySymbols(user)); // [Symbol(id)]

// Well-known symbols
const iterable = {
  [Symbol.iterator]: function* () {
    yield 1;
    yield 2;
    yield 3;
  }
};

Proxy & Reflect

A Proxy wraps an object and intercepts operations on it.

const handler = {
  get(target, prop) {
    if (prop in target) {
      return target[prop];
    }
    return `Property ${String(prop)} not found`;
  },
  set(target, prop, value) {
    if (typeof value !== 'number') {
      throw new TypeError('Must be a number');
    }
    target[prop] = value;
    return true;
  }
};

const nums = new Proxy({}, handler);
nums.a = 10;
console.log(nums.a);   // 10
console.log(nums.b);   // "Property b not found"
nums.c = 'hello';    // TypeError

WeakMap & WeakSet

WeakMap keys must be objects. They do not prevent garbage collection. 

const cache = new WeakMap();

function process(obj) {
  if (!cache.has(obj)) {
    const result = heavyCalculation(obj);
    cache.set(obj, result);
  }
  return cache.get(obj);
}

// When 'obj' is no longer referenced elsewhere, it can be garbage collected.
// The WeakMap entry disappears automatically.

Memoization Pattern

function memoize(fn) {
  const cache = new Map();
  return function(...args) {
    const key = JSON.stringify(args);
    if (cache.has(key)) {
      return cache.get(key);
    }
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

const fib = memoize(function(n) {
  if (n < 2) return n;
  return fib(n - 1) + fib(n - 2);
});

console.log(fib(50)); // Instant, thanks to memoization

Debounce & Throttle

// Debounce: run after user stops typing
function debounce(fn, delay) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

searchInput.addEventListener('input', debounce((e) => {
  fetchResults(e.target.value);
}, 300));

// Throttle: run at most once per interval
function throttle(fn, limit) {
  let inThrottle;
  return function(...args) {
    if (!inThrottle) {
      fn.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
}

window.addEventListener('scroll', throttle(() => {
  console.log('Scrolled!');
}, 200));
