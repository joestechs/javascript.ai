Project 1: To-Do App (DOM + Events)

Concepts: DOM manipulation, event delegation, localStorage. 

// Core structure
const form = document.querySelector('#todo-form');
const list = document.querySelector('#todo-list');
let todos = JSON.parse(localStorage.getItem('todos')) || [];

function render() {
  list.innerHTML = todos.map((todo, i) => `
    <li data-index="${i}">
      <span class="${todo.done ? 'done' : ''}">${todo.text}</span>
      <button class="toggle">✓</button>
      <button class="delete">✕</button>
    </li>
  `).join('');
  localStorage.setItem('todos', JSON.stringify(todos));
}

form.addEventListener('submit', (e) => {
  e.preventDefault();
  const input = form.querySelector('input');
  todos.push({ text: input.value, done: false });
  input.value = '';
  render();
});

list.addEventListener('click', (e) => {
  const index = e.target.closest('li')?.dataset.index;
  if (!index) return;

  if (e.target.matches('.toggle')) {
    todos[index].done = !todos[index].done;
  }
  if (e.target.matches('.delete')) {
    todos.splice(index, 1);
  }
  render();
});

render();

Project 3: Real-Time Chat (WebSockets)

Concepts: WebSocket API, event-driven architecture, JSON messaging. 

const ws = new WebSocket('wss://chat.example.com');

ws.onopen = () => console.log('Connected');

ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  appendMessage(msg.user, msg.text);
};

function sendMessage(text) {
  const payload = JSON.stringify({ user: 'Me', text });
  ws.send(payload);
}

Project 4: Search Autocomplete (Debounce + Fetch)

Concepts: Debounce, AbortController, dynamic suggestions. 

const searchInput = document.querySelector('#search');
let controller;

const debouncedSearch = debounce(async (query) => {
  if (controller) controller.abort();
  controller = new AbortController();

  try {
    const res = await fetch(`/api/search?q=${query}`, {
      signal: controller.signal
    });
    const results = await res.json();
    renderSuggestions(results);
  } catch (err) {
    if (err.name !== 'AbortError') console.error(err);
  }
}, 300);

searchInput.addEventListener('input', (e) => {
  debouncedSearch(e.target.value);
});

Project 5: Canvas Drawing App

Concepts: Canvas API, mouse events, state management. 

const canvas = document.querySelector('#draw');
const ctx = canvas.getContext('2d');
let drawing = false;

canvas.addEventListener('mousedown', () => drawing = true);
canvas.addEventListener('mouseup', () => drawing = false);
canvas.addEventListener('mousemove', (e) => {
  if (!drawing) return;
  ctx.lineTo(e.offsetX, e.offsetY);
  ctx.stroke();
});
