Chapter 08

Testing JavaScript

Testing proves your code works. It also documents how your code should be used.

Jest Basics

Jest is the most popular JavaScript testing framework. It works out of the box. 

// sum.js
export function sum(a, b) {
  return a + b;
}

// sum.test.js
import { sum } from './sum.js';

describe('sum', () => {
  test('adds 1 + 2 to equal 3', () => {
    expect(sum(1, 2)).toBe(3);
  });

  test('adds negative numbers', () => {
    expect(sum(-1, -1)).toBe(-2);
  });
});

Common Matchers

expect(value).toBe(4);           // Exact equality (===)
expect(value).toEqual({a: 1});   // Deep equality for objects
expect(value).toBeTruthy();      // Any truthy value
expect(value).toBeFalsy();       // Any falsy value
expect(value).toBeNull();        // null
expect(value).toBeUndefined();   // undefined
expect(value).toBeDefined();     // Not undefined
expect(arr).toContain('item'); // Array contains item
expect(str).toMatch(/hello/);  // Regex match
expect(fn).toThrow();            // Function throws error

Async Tests

// Return a promise
test('fetches user', () => {
  return fetchUser(1).then(data => {
    expect(data.name).toBe('Ada');
  });
});

// Use async/await (cleaner)
test('fetches user', async () => {
  const data = await fetchUser(1);
  expect(data.name).toBe('Ada');
});

// Expect a rejection
test('throws on bad id', async () => {
  await expect(fetchUser(-1)).rejects.toThrow('Invalid');
});

Mocking

// Mock a function
const mockFn = jest.fn();
mockFn('arg');
expect(mockFn).toHaveBeenCalledWith('arg');
expect(mockFn).toHaveBeenCalledTimes(1);

// Mock a module
jest.mock('./api.js', () => ({
  fetchUser: jest.fn(() => Promise.resolve({ name: 'Mock' }))
}));

// Spy on a method
const spy = jest.spyOn(console, 'log');
console.log('test');
expect(spy).toHaveBeenCalled();
spy.mockRestore(); // Clean up

Test Coverage

# Run tests with coverage report
npx jest --coverage

# Coverage metrics:
# - Statements: % of lines executed
# - Branches: % of if/else paths taken
# - Functions: % of functions called
# - Lines: % of lines covered
